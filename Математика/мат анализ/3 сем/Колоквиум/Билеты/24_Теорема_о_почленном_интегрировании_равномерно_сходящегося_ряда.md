<span style="background:#d3f8b6">Th:</span> Теорема о почленном интегрировании

$$\displaylines{
\text{Пусть} \\
1) \ \sum_{n=1}^{+\infty} u_{n} \ \text{- сходится равномерно на $[a, b]$ к $u(x)$} \\
2) \ \exists \int_{a}^{x}u_{n}(x)dx \ \forall n \in \mathbb{N} ,\ x\in[a, b] \\ \\
\text{Тогда} \\
a) \ \exists \int_{a}^{x}u(x)dx \\
b) \ \sum_{n=1}^{+\infty}\int_{a}^{x}u_{n}(x)dx = \int_{a}^{x}\sum_{n=1}^{+\infty} u_{n}(x)dx = \int_{a}^{x}u(x)dx
}$$
$$\displaylines{
\text{Докажем $\exists \int_{a}^{x}u(x)dx$} \\
\sum_{n=1}^{\infty} u_{n}(x) = S_{n}(x) + R_{n}(x) \ \text{- сходится равномерно}. \\
\forall \epsilon>0 \exists n_{0} \in \mathbb{N}: \forall x \in [a, b], \ \forall n > n_{0} \implies |R_{n}(x)| < \epsilon \\ \\
\text{Условие интегрируемости:} \\ 
\forall \epsilon > 0 \ \exists \delta>0: \forall P[a, b], \ \lambda < \delta \implies |\sum_{k=1}^{n}\omega_{k}^{u}\Delta x_{k}| < \epsilon; \\
u(x) = S_{n}(x) + R_{n}(x); \ S_{n}(x) \to \sum_{k=1}^{n}\omega_{k}^{S_{n}}\Delta x_{k} \\
R_{n}(x) \to \sum_{k=1}^{n} \omega_{k}^{R_{n}}\Delta x_{k} \\
\sum\omega_{k}^{u}\Delta x_{k} \leq \sum_{k=1}^{n}\omega_{k}^{S_{n}}\Delta x_{k} + \sum_{k=1}^{n} \omega_{k}^{R_{n}}\Delta x_{k} = \sigma_{1} + \sigma_{2}\\ \\
a) \ \text{Оценим $\omega_{k}^{S_{n}}$:} \ S_{n} = \sum_{k=1}^{n}u_{k}(x) \in J_{R}[a, b] \implies \sigma_{1} < \epsilon_{2}; \\
b) \ \text{Оценим $\omega_{k}^{R_{n}}$:} \  \ \omega_{k}^{R_{n}} = sup|R_{n}(x') - R_{n}(x'')| \leq sup|R_{n}(x')| + sup|R_{n}(x'')| \leq 2\epsilon_{2}; \\

\sigma_{2} = \sum_{k=1}^{n}\omega_{k}^{R_{n}}\Delta x_{k} \leq 2\epsilon_{1}\sum_{k=1}^{n}\Delta x_{k} = 2\epsilon_{1}(b-a) \\

\text{Возьмем:} \ \epsilon_{2} = \frac{\epsilon}{4}; \ \epsilon_{1} = \frac{\epsilon}{4(b-a)} \implies \sum_{k=1}^{n}\omega_{k}^{u}\Delta x_{k} < \epsilon; \\ \implies \exists \int_{a}^x u(x)dx
}$$

$$\displaylines{
\text{Докажем:} \ \sum_{n=1}^{+\infty} \int_{a}^{x}u_{n}(x)dx = \int_{a}^{x}\sum_{n=1}^{+\infty} u_{n}(x)dx = \int_{a}^{x}u(x)dx \\ \\

a) \ S_{n}(x) = \sum_{k=1}^{n}u_{k}(x) \in J_{R}[a, b], \ \text{т.е.} \int_{a}^{x} \left(\sum_{k=1}^{n}u_{k}(x)\right)dx = \sum_{k=1}^{n} \int_{a}^{x}u_{k}(x)dx \\
\exists \ \int_{a}^{x} u(x)dx = u^{*}(x) \\
|u^*(x) - S_{n}(x)| = \left |\int_{a}^x u(x)dx - \sum_{k=1}^n\int_{a}^xu_{k}(x)dx\right| = \left | u(x) = \sum_{k=1}^{+\infty}u_{k}(x) \right | = \\
= \left| \int_{a}^x\sum_{k=1}^{+\infty}u_{k}(x)dx - \int_{a}^x \sum_{k=1}^{n}u_{k}(x)dx\right| = \int_{a}^{x}\left | \sum_{k=1}^{+\infty}u_{k}(x) - \sum_{k=1}^{n}u_{k}(x)\right |dx \leq \\
\leq \int_{a}^x\left | \sum_{k=n+1}^xu_{k}(x) dx \right | = \int_{a}^x \left | R_{n}(x) \right |dx \leq \epsilon_{2} \frac{b - a}{2}; \ \epsilon_{2} = \frac{2\epsilon}{b - a}; \\
\text{Получили, что} \ |u^*(x) - S_{n}(x)| < \epsilon \implies \int_{a}^xu(x)dx=\sum_{k=1}^{+\infty}\int_{a}^xu_{k}(x)dx \ \ \blacksquare
}$$
