
$$\displaylines{
\text{Рассмотрим в пространстве метрику ф-ий $f(x)$} \\
\text{$\rho(f, f_{n}) = \sqrt{ \int_{a}^{b}(f - f_{n})dx }$ - квадратичное отклонение на $[a, b]$} \\ 
\text{Говорят, что $f_{n}(x) \to f(x)$ при $n \to +\infty$ , если $\sqrt{ \int_{a}^{b} (f - f_{n})dx } \to 0$ при $n \to +\infty$} \\
\text{$f_{n}(x) \rightrightarrows f(x)$ на $[a, b]$ при $n \to +\infty$, если $\sup|f - f_{n}| \to 0$ при $n \to +\infty$;}  \\
\text{равномерно сходится} \implies \text{сходится в среднем} \\
\text{Обратное не верно}
}$$
